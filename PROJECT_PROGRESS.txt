@app.post("/login")
async def do_login(login: str = Form(...), password: str = Form(...)):
    l = login.strip()
    u = load_users().get(l)
    if u and verify_pw(password.strip(), u["pw"]):
        r = RedirectResponse("/owner" if l == "admin" else "/account", 303)
        r.set_cookie("user", l, httponly=True, samesite="lax")
        return r
    return RedirectResponse("/login?err=1", 303)

async def do_reg(login: str = Form(...), password: str = Form(...)):
    users = load_users()

    USERS_FILE.write_text(json.dumps(users, indent=2))
    r = RedirectResponse("/account", 303)
    r.set_cookie("user", l, httponly=True, samesite="lax")
    return r

@app.post("/change-password")
async def ch_pass(request: Request, new_password: str = Form(...)):
    l = request.cookies.get("user", "")
    if l:

async def add_product(request: Request, name: str = Form(...), desc: str = Form(""), slug: str = Form(...), price: int = Form(...), photo: UploadFile = File(...)):
    if request.cookies.get("user") != "admin": return RedirectResponse("/login", 303)
    if check_stop_words(f"{name} {desc}"): return RedirectResponse("/", 303)
    fname = f"prod_{int(time.time())}_{photo.filename}"
    with open(f"/app/media/products/{fname}", "wb") as f: shutil.copyfileobj(photo.file, f)
    cat_name = "Субстраты почвенные" if slug == "substrates" else "Наборы и комплекты"
    save_product({"name": name, "desc": desc, "slug": slug, "cat": cat_name, "price": price, "img": f"/media/products/{fname}"})
    return RedirectResponse("/", 303)

@app.post("/owner/delete-product/{pid}")
async def del_prod(request: Request, pid: int):
    if request.cookies.get("user") == "admin":
        PROD_FILE.write_text(json.dumps([x for x in get_products() if x["id"] != pid], ensure_ascii=False, indent=2), encoding="utf-8")

@app.post("/owner/lecture")
async def save_lec(request: Request, topic: str = Form(...), lec_id: int = Form(...), title: str = Form(...), text: str = Form(""), photo: UploadFile = File(None)):
    if request.cookies.get("user") != "admin": return RedirectResponse("/login", 303)
    data = get_topics()
    img_url = ""
    if photo and photo.filename:
        fname = f"lec_{int(time.time())}_{photo.filename}"
        with open(f"/app/media/{fname}", "wb") as f: shutil.copyfileobj(photo.file, f)
        img_url = f"/media/{fname}"
    lecs = data[topic]["lectures"]
    found = next((x for x in lecs if x["id"] == lec_id), None)
    if found:
        found["title"], found["text"] = title, text
        if img_url: found["img"] = img_url
    else:
        lecs.append({"id": lec_id, "title": title, "text": text, "img": img_url})
    LEC_FILE.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
    return RedirectResponse(f"/encyclopedia/{topic}", 303)

@app.get("/logout")
async def logout():
    r = RedirectResponse("/", 303)
    r.delete_cookie("user")
    return r

import hashlib, os, hmac, json
from pathlib import Path

DATA_DIR = Path("/app/data")
DATA_DIR.mkdir(exist_ok=True)
USERS_FILE = DATA_DIR / "users.json"

def hash_pw(pw: str, salt: bytes | None = None) -> str:
    salt = salt or os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", pw.encode(), salt, 200_000)
    return f"{salt.hex()}:{dk.hex()}"

def verify_pw(pw: str, stored: str) -> bool:
    if ":" not in stored:
        return hmac.compare_digest(pw, stored)
    s_hex, h_hex = stored.split(":", 1)
    dk = hashlib.pbkdf2_hmac("sha256", pw.encode(), bytes.fromhex(s_hex), 200_000)
    return hmac.compare_digest(dk.hex(), h_hex)

def load_users() -> dict:
    if not USERS_FILE.exists():
        u = {"admin": {"pw": hash_pw("admin"), "2fa": False}}
        USERS_FILE.write_text(json.dumps(u, indent=2))
        return u
    return json.loads(USERS_FILE.read_text())

]

def get_products() -> list:
    if not PROD_FILE.exists():
        PROD_FILE.write_text(json.dumps(INIT_PRODUCTS, ensure_ascii=False, indent=2), encoding="utf-8")
    return json.loads(PROD_FILE.read_text(encoding="utf-8"))

def save_product(item: dict):
    items = get_products()
    item["id"] = max((x["id"] for x in items), default=0) + 1
    items.append(item)
    PROD_FILE.write_text(json.dumps(items, ensure_ascii=False, indent=2), encoding="utf-8")

def check_stop_words(text: str) -> list[str]:
    low = text.lower()
    return [w for w in STOP_WORDS if w in low]
