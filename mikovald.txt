def is_ready(keys: list[str]) -> bool:
    p = Path("/app/integrations.txt")
    if not p.exists():
        return False
    cfg = dict(line.split("=", 1) for line in p.read_text().splitlines() if "=" in line)
    return all(bool(cfg.get(k, "").strip()) for k in keys)

@app.get("/", response_class=HTMLResponse)
async def index(request: Request, q: str = "", cat: str = ""):
    email = request.cookies.get("user_email", "")
    disc = 0.95 if email else 1.0
    items = [
        {**p, "final": int(p["price"] * disc)} for p in PRODUCTS
        if (not q or q.lower() in p["name"].lower()) and (not cat or cat == p["cat"])
    ]
    return templates.TemplateResponse("index.html", {
        "request": request, "products": items, "email": email, "q": q, "cat": cat,
        "yk": is_ready(["YOOKASSA_SHOP_ID", "YOOKASSA_SECRET_KEY"]),
        "pr": is_ready(["POCHTA_WIDGET_ID"]),
    })

@app.post("/login")
async def login(email: str = Form(...)):
    r = RedirectResponse("/", 303)
    r.set_cookie("user_email", email)
    return r

@app.get("/logout")
async def logout():
    r = RedirectResponse("/", 303)
    r.delete_cookie("user_email")
    return r
EOF

def is_ready(keys: list[str]) -> bool:
    p = Path("/app/integrations.txt")
    if not p.exists():
        return False
    cfg = dict(line.split("=", 1) for line in p.read_text().splitlines() if "=" in line)

@app.get("/", response_class=HTMLResponse)
async def index(request: Request, q: str = "", cat: str = ""):
    user = request.cookies.get("user_email", "")
    is_admin = request.cookies.get("is_admin") == "1"
    disc = 0.95 if user else 1.0
    items = [
        {**p, "final": int(round(p["price"] * disc))} for p in PRODUCTS
        if (not q or q.lower() in p["name"].lower()) and (not cat or cat == p["cat"])
    ]
    return templates.TemplateResponse(request, "index.html", {
        "request": request, "products": items, "email": user, "is_admin": is_admin, "q": q, "cat": cat,

        "pr": is_ready(["POCHTA_WIDGET_ID"]),
    })

@app.post("/login")
async def login(email: str = Form(...), password: str = Form("")):
    r = RedirectResponse("/", 303)
    if email.strip() == "admin" and password.strip() == "admin":
        r.set_cookie("user_email", "admin")
        r.set_cookie("is_admin", "1")
    else:
        r.set_cookie("user_email", email.strip())
        r.delete_cookie("is_admin")
    return r

@app.get("/logout")
async def logout():
    r = RedirectResponse("/", 303)
    r.delete_cookie("user_email")
    r.delete_cookie("is_admin")

def ctx(request: Request, **extra):
    user = request.cookies.get("user", "")
    is_admin = request.cookies.get("admin") == "1"
    return {"request": request, "user": user, "is_admin": is_admin, "disc": 0.95 if user else 1.0, **extra}

@app.get("/", response_class=HTMLResponse)
@app.get("/category/{slug}", response_class=HTMLResponse)
async def home(request: Request, slug: str = "", q: str = ""):
    c = ctx(request, slug=slug, q=q)
    items = [{**p, "final": int(round(p["price"] * c["disc"]))} for p in PRODUCTS if (not slug or p["slug"] == slug) and (not q or q.lower() in p["name"].lower())]
    return templates.TemplateResponse(request, "home.html", {**c, "products": items})

@app.get("/owner", response_class=HTMLResponse)
async def owner_page(request: Request):
    c = ctx(request)
    if not c["is_admin"]:
        return RedirectResponse("/login", 303)
    return templates.TemplateResponse(request, "owner.html", c)

@app.post("/login")
async def login_post(login: str = Form(...), password: str = Form("")):
    is_adm = (login.strip() == "admin" and password.strip() == "admin")
    r = RedirectResponse("/owner" if is_adm else "/", 303)
    r.set_cookie("user", login.strip())
    r.set_cookie("admin", "1" if is_adm else "0")
    return r

@app.get("/logout")
async def logout():
    r = RedirectResponse("/", 303)
    r.delete_cookie("user")
    r.delete_cookie("admin")
    return r

import hashlib, os, hmac, json
from pathlib import Path

DATA_DIR = Path("/app/data")
DATA_DIR.mkdir(exist_ok=True)
USERS_FILE = DATA_DIR / "users.json"

def hash_pw(pw: str, salt: bytes | None = None) -> str:
    salt = salt or os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", pw.encode(), salt, 200_000)
    return f"{salt.hex()}:{dk.hex()}"

def verify_pw(pw: str, stored: str) -> bool:
    if ":" not in stored:
        return hmac.compare_digest(pw, stored)
    s_hex, h_hex = stored.split(":", 1)
    dk = hashlib.pbkdf2_hmac("sha256", pw.encode(), bytes.fromhex(s_hex), 200_000)
    return hmac.compare_digest(dk.hex(), h_hex)

def load_users() -> dict:
    if not USERS_FILE.exists():
        u = {"admin": {"pw": hash_pw("admin"), "2fa": False}}
        USERS_FILE.write_text(json.dumps(u, indent=2))
        return u
    return json.loads(USERS_FILE.read_text())
